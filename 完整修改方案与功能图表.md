```python
# algorithms/enhanced_communication.py
def calculate_communication_quality(p1, p2, terrain_type, freq_ghz=2.4, weather='clear'):
    """
    计算通信质量（0-1）
    
    公式: Quality = 1 / (1 + exp((TotalLoss - Threshold)/10))
    
    TotalLoss = Friis损耗 + 地形衰减 + 大气衰减
    """
    distance_km = distance(p1, p2) / 1000
    
    # Friis自由空间损耗
    fspl = 20 * math.log10(distance_km) + 20 * math.log10(freq_ghz) + 92.45
    
    # 地形衰减系数 (dB)
    terrain_attenuation = {
        'plain': 0, 'hill': 3, 'mountain': 10, 
        'forest': 5, 'urban': 8
    }[terrain_type]
    
    # 大气衰减 (ITU-R P.676简化模型)
    atmospheric_atten = 0.02 * distance_km if weather == 'clear' else 0.1 * distance_km
    
    total_loss = fspl + terrain_attenuation + atmospheric_atten
    
    # Sigmoid转换到0-1质量
    max_acceptable_loss = 120  # dB
    quality = 1 / (1 + math.exp((total_loss - max_acceptable_loss) / 10))
    
    return quality
```

# algorithms/redundancy_evaluation.py
class RedundancyEvaluator:
    """链路冗余性评估器"""
    
    def calculate_node_importance_entropy(self, nodes, chains):
        """
        基于熵权法的节点重要性评估
        
        指标:
        1. 度数中心性 - 连接的目标数
        2. 介数中心性 - 经过的链数  
        3. 瓶颈程度 - 不可替代性
        4. OODA环节关键度 - 侦察/指挥/火力的环节权重
        """
        indicators = ['degree', 'betweenness', 'bottleneck', 'ooda_criticality']
        
        # 构建判断矩阵
        indicator_matrix = self._build_indicator_matrix(nodes, chains)
        
        # 计算熵值确定权重
        weights = self._entropy_weight_method(indicator_matrix)
        
        # 综合评分
        importance_scores = {}
        for node_id in nodes:
            score = sum(w * indicator_matrix[node_id][i] 
                       for i, w in enumerate(weights))
            importance_scores[node_id] = score
            
        return importance_scores
    
    def _entropy_weight_method(self, matrix):
        """熵权法计算指标权重"""
        # 数据标准化
        normalized = self._normalize_matrix(matrix)
        
        # 计算信息熵
        entropy = []
        for j in range(len(normalized[0])):
            p = normalized[:, j] / np.sum(normalized[:, j])
            e = -np.sum(p * np.log(p + 1e-10))
            entropy.append(e)
        
        # 计算权重 (熵值越小，权重越大)
        weights = [(1 - e) / sum(1 - e for e in entropy) for e in entropy]
        return weights
    
    def generate_replacement_strategy(self, failed_node, current_chains, spare_chains):
        """
        生成接替策略
        
        策略:
        1. 优先选择冗余度低（节点差异大）的替代链
        2. 保持OODA环节完整性
        3. 最小化对现有链的影响
        """
        candidates = []
        for chain in spare_chains:
            redundancy = self._calculate_redundancy(current_chains, chain)
            ooda_integrity = self._check_ooda_integrity(chain, failed_node)
            impact = self._calculate_replacement_impact(current_chains, chain)
            
            candidates.append({
                'chain': chain,
                'redundancy': redundancy,
                'ooda_integrity': ooda_integrity,
                'impact': impact,
                'score': redundancy * 0.4 + ooda_integrity * 0.4 + (1-impact) * 0.2
            })
        
        # 按综合得分排序
        candidates.sort(key=lambda x: x['score'], reverse=True)
        return candidates[0] if candidates else None
```

# algorithms/dynamic_deployment.py
class TerrainGrid:
    """栅格化地形模型"""
    
    def __init__(self, bounds, grid_size=100):
        self.grid_size = grid_size  # 米
        self.bounds = bounds  # (x_min, x_max, y_min, y_max)
        self.grids = self._initialize_grids()
        
    def _initialize_grids(self):
        """初始化网格"""
        grids = {}
        x_range = np.arange(self.bounds[0], self.bounds[1], self.grid_size/1000)
        y_range = np.arange(self.bounds[2], self.bounds[3], self.grid_size/1000)
        
        for i, x in enumerate(x_range):
            for j, y in enumerate(y_range):
                grid_id = f"{i}_{j}"
                grids[grid_id] = {
                    'center': (x + self.grid_size/2000, y + self.grid_size/2000),
                    'terrain_type': self._get_terrain_type(x, y),
                    'threat_level': self._get_threat_level(x, y),
                    'reachability_score': 0,
                    'mobility_cost': 0
                }
        return grids
    
    def calculate_grid_scores(self, targets, current_nodes):
        """计算每个网格的得分"""
        for grid_id, grid in self.grids.items():
            # 可达性得分
            reachability = self._calculate_reachability(grid['center'], targets)
            
            # 威胁得分
            threat = grid['threat_level']
            
            # 机动消耗
            mobility = self._calculate_mobility_cost(
                grid['center'], current_nodes, grid['terrain_type']
            )
            
            grid['reachability_score'] = reachability
            grid['threat_penalty'] = threat
            grid['mobility_cost'] = mobility
            grid['total_score'] = reachability - threat * 0.3 - mobility * 0.2


class AWPSODeploymentOptimizer:
    """自适应权重粒子群优化部署"""
    
    def __init__(self, terrain_grid, nodes, targets):
        self.terrain_grid = terrain_grid
        self.nodes = nodes
        self.targets = targets
        
    def optimize(self, max_iterations=100, num_particles=50):
        """
        AWPSO优化
        
        目标: max(可达性 - 威胁 - 机动消耗)
        
        自适应权重调整:
        - 早期: 大权重，全局搜索
        - 中期: 平衡权重
        - 后期: 小权重，局部精细搜索
        """
        # 初始化粒子群
        particles = self._initialize_particles(num_particles)
        velocities = [np.zeros(2) for _ in range(num_particles)]
        
        personal_best = particles.copy()
        global_best = self._find_best(particles)
        
        for iteration in range(max_iterations):
            # 自适应惯性权重
            w = self._adaptive_inertia_weight(iteration, max_iterations)
            
            # 自适应学习因子
            c1, c2 = self._adaptive_learning_factors(iteration, max_iterations)
            
            for i, particle in enumerate(particles):
                # 更新速度
                r1, r2 = np.random.random(2)
                velocities[i] = (
                    w * velocities[i] +
                    c1 * r1 * (personal_best[i] - particle) +
                    c2 * r2 * (global_best - particle)
                )
                
                # 更新位置
                particles[i] = particle + velocities[i]
                
                # 边界处理
                particles[i] = self._constrain_to_valid_grids(particles[i])
                
                # 更新个人最优
                if self._fitness(particles[i]) > self._fitness(personal_best[i]):
                    personal_best[i] = particles[i]
            
            # 更新全局最优
            global_best = self._find_best(particles)
            
            # 早停检查
            if self._convergence_check(particles):
                break
        
        return global_best
    
    def _adaptive_inertia_weight(self, iteration, max_iterations):
        """自适应惯性权重"""
        # 非线性递减
        progress = iteration / max_iterations
        w_max, w_min = 0.9, 0.4
        return w_max - (w_max - w_min) * (progress ** 2)
    
    def _adaptive_learning_factors(self, iteration, max_iterations):
        """自适应学习因子"""
        progress = iteration / max_iterations
        # 早期强调个体经验，后期强调群体经验
        c1 = 2.5 - 1.5 * progress
        c2 = 0.5 + 1.5 * progress
        return c1, c2
```
