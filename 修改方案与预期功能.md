from dataclasses import dataclass

class NodeType(Enum):
    RECON = "recon"
    COMMAND = "command"
    FIRE = "fire"
    TARGET = "target"

@dataclass
class Node:
    """节点基础数据类"""
    node_id: str
    node_type: NodeType
    name: str
    coord: Tuple[float, float, float]  # x, y, h
    velocity: Tuple[float, float, float]  # vx, vy, vh
    
    def distance_to(self, other: 'Node') -> float:
        """计算与另一节点的距离"""
        return ((self.coord[0] - other.coord[0])**2 +

    parallel_limit: int
    recon_range: float
    recon_precision: float

### 3.1 增强型SSL构建算法 (algorithms/enhanced_ssl.py)

```python
"""
增强型SSL构建算法 - 基于论文第3章

核心改进：
1. 引入时序约束建模
2. 多目标优化框架
3. 动态重构能力
"""

import numpy as np
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass
from enum import Enum

class ConstraintType(Enum):
    TEMPORAL = "temporal"      # 时序约束
    SPATIAL = "spatial"        # 空间约束
    CAPACITY = "capacity"      # 容量约束
    RESOURCE = "resource"      # 资源约束

@dataclass
class TemporalConstraint:
    """时序约束"""
    node_id: str
    earliest_start: float      # 最早开始时间
    latest_finish: float       # 最晚完成时间
    duration: float            # 持续时间

@dataclass
class SSLProblem:
    """SSL问题定义"""
    targets: Dict[str, dict]
    recon_nodes: Dict[str, dict]
    command_nodes: Dict[str, dict]
    fire_nodes: Dict[str, dict]
    
    # 约束条件
    temporal_constraints: Dict[str, TemporalConstraint]
    spatial_constraints: List[Tuple[str, str, float]]  # (node1, node2, max_distance)
    
    # 优化目标权重
    objectives: Dict[str, float]  # reaction_time, precision, cost, coverage

class EnhancedSSLBuilder:
    """增强型SSL构建器"""
    
    def __init__(self, problem: SSLProblem):
        self.problem = problem
        self.feasible_chains = []
        self.pareto_front = []
        
    def build_with_temporal_constraints(self) -> List[KillChain]:
        """
        构建满足时序约束的杀伤链
        
        步骤：
        1. 生成所有可行链（考虑空间约束）
        2. 筛选满足时序约束的链
        3. 多目标优化选择最优链
        """
        # 步骤1: 生成候选链
        candidates = self._generate_candidate_chains()
        
        # 步骤2: 时序可行性检查
        feasible = self._filter_temporal_feasible(candidates)
        
        # 步骤3: 多目标优化
        self.pareto_front = self._multi_objective_optimize(feasible)
        
        return self.pareto_front
    
    def _generate_candidate_chains(self) -> List[dict]:
        """生成所有满足空间约束的候选链"""
        candidates = []
        for target_id, target in self.problem.targets.items():
            for recon_id, recon in self.problem.recon_nodes.items():
                if not self._is_within_range(target, recon, recon['range']):
                    continue
                for cmd_id, cmd in self.problem.command_nodes.items():
                    if not self._can_communicate(recon, cmd, cmd['comm_distance']):
                        continue
                    for fire_id, fire in self.problem.fire_nodes.items():
                        if not self._can_communicate(cmd, fire, fire['comm_distance']):
                            continue
                        if not self._is_within_range(fire, target, fire['range']):
                            continue
                        
                        candidates.append({
                            'target_id': target_id,
                            'recon_id': recon_id,
                            'command_id': cmd_id,
                            'fire_id': fire_id,
                            'metrics': self._calculate_metrics(target, recon, cmd, fire)
                        })
        return candidates
    
    def _filter_temporal_feasible(self, candidates: List[dict]) -> List[dict]:
        """筛选满足时序约束的候选链"""
        feasible = []
        for chain in candidates:
            # 计算各阶段时间
            detection_time = self._calculate_detection_time(chain)
            decision_time = self._calculate_decision_time(chain)
            strike_time = self._calculate_strike_time(chain)
            
            # 检查时序约束
            if self._check_temporal_feasibility(
                chain['target_id'], 
                detection_time, 
                decision_time, 
                strike_time
            ):
                chain['temporal'] = {
                    'detection_time': detection_time,
                    'decision_time': decision_time,
                    'strike_time': strike_time,
                    'total_time': detection_time + decision_time + strike_time
                }
                feasible.append(chain)
        return feasible
    
    def _multi_objective_optimize(self, feasible: List[dict]) -> List[KillChain]:
        """
        多目标优化 - 基于NSGA-II框架
        
        目标函数：
        f1 = 最小化反应时间
        f2 = 最大化打击精度 (即最小化σ²)
        f3 = 最小化成本
        f4 = 最大化威胁覆盖
        """
        if not feasible:
            return []
        
        # 归一化目标值
        objectives = np.array([
            [c['metrics']['reaction_time'],
             c['metrics']['precision'],
             c['metrics']['cost'],
             -c['metrics']['threat_coverage']]  # 负号因为最大化
            for c in feasible
        ])
        
        # 非支配排序
        fronts = self._non_dominated_sort(objectives)
        
        # 构建帕累托前沿
        pareto_chains = []
        for front_idx, front in enumerate(fronts[:3], 1):  # 取前3层
            for idx in front:
                chain_data = feasible[idx]
                chain = KillChain(
                    chain_id=f"chain_{len(pareto_chains)}",
                    target_id=chain_data['target_id'],
                    recon_id=chain_data['recon_id'],
                    command_id=chain_data['command_id'],
                    fire_id=chain_data['fire_id'],
                    detection_time=chain_data['temporal']['detection_time'],
                    decision_time=chain_data['temporal']['decision_time'],
                    strike_time=chain_data['temporal']['strike_time'],
                    reaction_time=chain_data['metrics']['reaction_time'],
                    precision=chain_data['metrics']['precision'],
                    fire_power=chain_data['metrics']['fire_power'],
                    cost=chain_data['metrics']['cost'],
                    match_score=chain_data['metrics']['match_score']
                )
                chain.pareto_layer = front_idx
                chain.is_pareto_optimal = (front_idx == 1)
                pareto_chains.append(chain)
        
        return pareto_chains
    
    def _non_dominated_sort(self, objectives: np.ndarray) -> List[List[int]]:
        """非支配排序"""
        n = len(objectives)
        domination_count = [0] * n
        dominated_solutions = [[] for _ in range(n)]
        fronts = [[]]
        
        for i in range(n):
            for j in range(i + 1, n):
                if self._dominates(objectives[i], objectives[j]):
                    dominated_solutions[i].append(j)
                    domination_count[j] += 1
                elif self._dominates(objectives[j], objectives[i]):
                    dominated_solutions[j].append(i)
                    domination_count[i] += 1
            
            if domination_count[i] == 0:
                fronts[0].append(i)
        
        i = 0
        while len(fronts[i]) > 0:
            next_front = []
            for p in fronts[i]:
                for q in dominated_solutions[p]:
                    domination_count[q] -= 1
                    if domination_count[q] == 0:
                        next_front.append(q)
            i += 1
            fronts.append(next_front)
        
        return fronts[:-1]  # 去除空的前沿
    
    def _dominates(self, a: np.ndarray, b: np.ndarray) -> bool:
        """判断a是否支配b"""
        return np.all(a <= b) and np.any(a < b)
```

### 3.2 动态重构算法 (algorithms/dynamic_reconfig.py)

```python
"""
动态重构算法 - 基于论文第4章

核心功能：
1. 节点失效检测
2. 快速重构策略
3. 性能保持优化
"""

import numpy as np
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass
from enum import Enum

class ReconfigTrigger(Enum):
    NODE_FAILURE = "node_failure"      # 节点失效
    PERFORMANCE_DEGRADE = "performance_degrade"  # 性能下降
    NEW_TARGET = "new_target"          # 新目标出现
    RESOURCE_CHANGE = "resource_change" # 资源变化

@dataclass
class ReconfigEvent:
    """重构事件"""
    event_type: ReconfigTrigger
    affected_nodes: List[str]
    timestamp: float
    severity: float  # 严重程度 0-1

class DynamicReconfigurator:
    """动态重构器"""
    
    def __init__(self, ssl_builder: EnhancedSSLBuilder):
        self.ssl_builder = ssl_builder
        self.current_chains = []
        self.reconfig_history = []
        
    def handle_event(self, event: ReconfigEvent) -> List[KillChain]:
        """
        处理重构事件
        
        策略：
        1. 局部重构：仅替换失效节点
        2. 全局重构：重新优化整个方案
        """
        if event.severity < 0.3:
            # 轻微影响，局部重构
            return self._local_reconfig(event)
        else:
            # 严重影响，全局重构
            return self._global_reconfig(event)
    
    def _local_reconfig(self, event: ReconfigEvent) -> List[KillChain]:
        """局部重构策略"""
        new_chains = []
        for chain in self.current_chains:
            if any(node_id in event.affected_nodes for node_id in 
                   [chain.recon_id, chain.command_id, chain.fire_id]):
                # 找到替代节点
                replacement = self._find_replacement(chain, event.affected_nodes)
                if replacement:
                    new_chain = self._apply_replacement(chain, replacement)
                    new_chains.append(new_chain)
                else:
                    # 无法局部重构，标记为失效
                    chain.is_active = False
                    new_chains.append(chain)
            else:
                new_chains.append(chain)
        
        self.current_chains = new_chains
        self.reconfig_history.append({
            'event': event,
            'strategy': 'local',
            'timestamp': event.timestamp
        })
        
        return new_chains
    
    def _global_reconfig(self, event: ReconfigEvent) -> List[KillChain]:
        """全局重构策略"""
        # 更新问题定义
        self.ssl_builder.problem = self._update_problem(event)
        
        # 重新构建
        new_chains = self.ssl_builder.build_with_temporal_constraints()
        
        self.current_chains = new_chains
        self.reconfig_history.append({
            'event': event,
            'strategy': 'global',
            'timestamp': event.timestamp
        })
        
        return new_chains
    
    def _find_replacement(self, chain: KillChain, affected_nodes: List[str]) -> Optional[dict]:
        """寻找替代节点"""
        # 根据失效节点类型寻找替代
        for node_id in affected_nodes:
            if node_id == chain.recon_id:
                return self._find_recon_replacement(chain)
            elif node_id == chain.command_id:
                return self._find_command_replacement(chain)
            elif node_id == chain.fire_id:
                return self._find_fire_replacement(chain)
        return None
    
    def evaluate_reconfig_impact(self) -> dict:
        """评估重构影响"""
        if len(self.reconfig_history) < 2:
            return {'impact': 0, 'stability': 1.0}
        
        # 计算性能变化
        recent = self.reconfig_history[-5:]  # 最近5次
        reaction_times = [r.get('avg_reaction_time', 0) for r in recent]
        
        return {
            'impact': np.std(reaction_times) if len(reaction_times) > 1 else 0,
            'stability': 1.0 - len(self.reconfig_history) / 100,  # 重构次数越多稳定性越低
            'reconfig_count': len(self.reconfig_history)
        }
```

    pareto_layer INT DEFAULT 0,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (scheme_id) REFERENCES scheme_master(scheme_id)
);

-- 目标-链映射表
CREATE TABLE target_chain_mapping (
    id INT AUTO_INCREMENT PRIMARY KEY,
    allocation_id VARCHAR(50) NOT NULL,
    target_id VARCHAR(50) NOT NULL,
    chain_id VARCHAR(50) NOT NULL,
    
    FOREIGN KEY (allocation_id) REFERENCES multi_target_allocations(allocation_id),
    FOREIGN KEY (target_id) REFERENCES target_nodes(node_id),
    FOREIGN KEY (chain_id) REFERENCES kill_chains(chain_id)

CREATE TABLE reconfig_history (
    reconfig_id INT AUTO_INCREMENT PRIMARY KEY,

    affected_nodes TEXT,  -- JSON格式存储节点列表
    strategy VARCHAR(20),  -- local/global
    severity FLOAT,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (scheme_id) REFERENCES scheme_master(scheme_id)
);

-- 评估结果表
CREATE TABLE evaluation_results (
    eval_id INT AUTO_INCREMENT PRIMARY KEY,
    scheme_id VARCHAR(50) NOT NULL,
    allocation_id VARCHAR(50),
    
    -- 评估指标
    timeliness_score FLOAT,
    accuracy_score FLOAT,
    firepower_score FLOAT,
    economy_score FLOAT,
    adaptability_score FLOAT,
    overall_score FLOAT,
    
    -- 评估详情(JSON)
    details TEXT,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (scheme_id) REFERENCES scheme_master(scheme_id),
    FOREIGN KEY (allocation_id) REFERENCES multi_target_allocations(allocation_id)
);

# services/integration_service.py

class SSLIntegrationService:
    """
    SSL系统集成服务
    
    整合所有功能模块，提供统一接口
    """
    
    def __init__(self):
        self.scheme_service = SchemeService()
        self.node_service = NodeService()
        self.algorithm_service = AlgorithmService()
        self.evaluation_service = EvaluationService()
        self.reconfig_service = ReconfigurationService()
        
    def create_scheme(self, scheme_data: dict) -> str:
        """创建新方案"""
        scheme_id = self.scheme_service.create(scheme_data)
        return scheme_id
    
    def build_ssl(self, scheme_id: str, algorithm: str = "enhanced") -> dict:
        """
        构建SSL杀伤链
        
        Args:
            scheme_id: 方案ID
            algorithm: 算法类型 (enhanced/mopso/pso/loop)
        
        Returns:
            构建结果，包含杀伤链列表和评估指标
        """
        # 获取方案数据
        scheme = self.scheme_service.get(scheme_id)
        nodes = self.node_service.get_by_scheme(scheme_id)
        
        # 执行算法
        if algorithm == "enhanced":
            result = self.algorithm_service.build_enhanced(scheme, nodes)
        elif algorithm == "mopso":
            result = self.algorithm_service.build_mopso(scheme, nodes)
